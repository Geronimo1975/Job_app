# Mobile UI (`/m/`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a mobile web UI at `/m/` in the existing Django dashboard (`dashboard/jobs`), recreating the 9 screens from the `mobile_app/` design handoff, reusing the existing bridge modules (`generation_bridge`, `search_bridge`, `sheet_bridge`) — no pipeline logic duplicated.

**Architecture:** New `dashboard/jobs/mobile_views.py` module with view functions that call the SAME bridge functions as the desktop views but render templates under `dashboard/jobs/templates/jobs/mobile/`, all extending a new `mobile/base.html` (tab bar + CSS tokens). New URLs live under the `/m/` prefix in `dashboard/jobs/urls.py`. Two new small modules: `dashboard/jobs/pipeline_status.py` (reads agent logs / rate-limit file, read-only) and `dashboard/jobs/settings_bridge.py` (reads/writes `config/settings.yaml` with `.bak` backup). `LoginRequiredMiddleware` (Django 6) is scoped to `/m/` only via a dedicated middleware class that checks `request.path`.

**Tech Stack:** Django 5/6 templates, HTMX 1.9 (vendored as a static file, no CDN dependency), hand-written CSS following the design tokens table from the spec. No new Python dependencies beyond what's already in `requirements.txt`.

## Global Constraints

- Reuse existing bridge functions (`generation_bridge.generate_package/send_package/already_sent_in_sheet/regenerate_pdfs`, `search_bridge.*`, `sheet_bridge.*`) — do not duplicate their logic in mobile views.
- Desktop routes (`/`, `/job/...`, `/cauta/...`, `/trimise/...`) and their templates (`jobs/*.html`, not `jobs/mobile/*.html`) must remain byte-for-byte unchanged.
- No schema migration: `send_state` gains the convention value `"applied"` but stays a `CharField` — no `choices` enforcement is added (existing field has none).
- Dark theme only (default from spec's Design Tokens table) — Light/True Black theme switch is UI-only chrome, not implemented this phase (Settings screen shows the "Appearance" row as a static label, not a working toggle) — this is a deliberate simplification, call it out in the Settings task.
- Pipeline screen is READ-ONLY — no "Run full pipeline" trigger.
- Settings screen READS and WRITES `config/settings.yaml`, with a `.bak` backup written before every save. Market "enabled" toggles write a new `enabled: true/false` key per market in YAML; the scraper (`scrapers/base.py::Market.from_config`) does NOT currently read this key, so toggling a market off in this phase has **no effect on the next scrape run** — document this limitation directly in the Settings template (small note text) and in the task below. Wiring `Market.from_config` to respect `enabled` is out of scope for this plan.
- Auth: `LoginRequiredMiddleware` applies only to paths starting with `/m/`. `/` (desktop) stays open, unchanged.
- Every mobile screen extends `jobs/mobile/base.html`; no CSS duplication across screen templates — all tokens/utility classes live in one stylesheet.

```python
"""Views pentru UI-ul mobil (/m/) — reutilizează bridge-urile desktop-ului,
doar template-ul și navigarea sunt diferite."""
import math


def score_ring_ctx(score: int | None, size: int, stroke: int) -> dict:
    """Context precalculat pentru `_score_ring.html` (Django templates nu fac
    aritmetică pe float ușor, deci geometria cercului se calculează aici)."""
    score = max(0, min(100, score or 0))
    half = size / 2
    radius = half - stroke / 2 - 1
    circumference = 2 * math.pi * radius
    arc = circumference * score / 100
    return {
        "score": score,
        "half": half,
        "radius": radius,
        "stroke_width": stroke,
        "arc": arc,
        "circumference": circumference,
        "num_size": round(size * 0.27),
    }
```

```python
from django.contrib import messages
from django.http import HttpResponseNotAllowed
from django.shortcuts import get_object_or_404, redirect, render

from .models import Job


def job_route(job) -> str:
    """`email` dacă jobul are Kontakt Email, altfel `portal` — regula din
    col AA „Bewerbungsart" a Sheet-ului, dedusă aici pentru UI."""
    return "email" if (job.email_hr or "").strip() else "portal"


_INBOX_FILTERS = {
    "draft": "draft",
    "ready": "generated",
    "sent": "sent",
    "error": "error",
}


def m_index(request):
    jobs = Job.objects.all()
    active_filter = request.GET.get("filter", "")
    if active_filter in _INBOX_FILTERS:
        jobs = jobs.filter(send_state=_INBOX_FILTERS[active_filter])

    counts = {
        key: Job.objects.filter(send_state=state).count()
        for key, state in _INBOX_FILTERS.items()
    }
    job_cards = [
        {"job": job, "route": job_route(job),
         "ring": score_ring_ctx(job.match_score, size=44, stroke=4)}
        for job in jobs
    ]
    return render(request, "jobs/mobile/inbox.html", {
        "job_cards": job_cards,
        "active_filter": active_filter,
        "counts": counts,
        "total": Job.objects.count(),
        "shown": len(job_cards),
        "active_tab": "inbox",
    })


def mark_applied(request, pk):
    if request.method != "POST":
        return HttpResponseNotAllowed(["POST"])
    job = get_object_or_404(Job, pk=pk)
    if job.send_state in ("draft", "error") and job_route(job) == "portal":
        job.send_state = "applied"
        job.save(update_fields=["send_state"])
        messages.success(request, "Marcat ca aplicat prin portal.")
    else:
        messages.error(request, "Doar joburile fără email (rută portal) pot fi marcate manual.")
    return redirect("m_job_detail", pk=pk)
```

```python
from django.utils import timezone

from . import generation_bridge, sheet_bridge


def _attachment_entries(job) -> list[dict]:
    from pathlib import Path
    return [
        {"idx": i, "name": Path(p).name, "exists": Path(p).exists()}
        for i, p in enumerate(job.attachments_json or [])
    ]


def m_execute_job(request, pk):
    if request.method != "POST":
        return HttpResponseNotAllowed(["POST"])
    job = get_object_or_404(Job, pk=pk)
    if job.send_state == "sent":
        messages.error(request, "Jobul e deja trimis — nu regenerez peste un pachet trimis.")
        return redirect("m_job_detail", pk=pk)
    res = generation_bridge.generate_package(job)
    if res["ok"]:
        job.email_subject = res["email_subject"]
        job.email_body = res["email_body"]
        job.anschreiben_text = res["anschreiben_text"]
        job.anschreiben_pdf_path = res["anschreiben_pdf_path"]
        job.bewerbung_pdf_path = res["bewerbung_pdf_path"]
        job.attachments_json = res["attachments"]
        job.send_state = "generated"
        job.generated_at = timezone.now()
        job.send_error = ""
        job.save()
        messages.success(request, "Pachet generat — verifică-l în Preview.")
        return redirect("m_preview_job", pk=pk)
    job.send_state = "error"
    job.send_error = res["error"]
    job.save(update_fields=["send_state", "send_error"])
    messages.error(request, f"Generare eșuată: {res['error']}")
    return redirect("m_job_detail", pk=pk)


def m_preview_job(request, pk):
    job = get_object_or_404(Job, pk=pk)
    if request.method == "POST":
        if job.send_state == "sent":
            messages.error(request, "Pachetul e deja trimis — arhiva nu se editează.")
            return redirect("m_preview_job", pk=pk)
        job.email_subject = request.POST.get("email_subject", job.email_subject)
        job.email_body = request.POST.get("email_body", job.email_body)
        new_text = request.POST.get("anschreiben_text", job.anschreiben_text)
        text_changed = new_text != job.anschreiben_text
        job.anschreiben_text = new_text
        job.save(update_fields=["email_subject", "email_body", "anschreiben_text"])
        if text_changed and job.anschreiben_pdf_path:
            res = generation_bridge.regenerate_pdfs(job)
            if res["ok"]:
                messages.success(request, "Editări salvate; PDF-urile refăcute.")
            else:
                messages.warning(request, f"Editări salvate, dar PDF-ul NU s-a refăcut: {res['error']}")
        else:
            messages.success(request, "Editări salvate.")
        if request.POST.get("next") == "transmit":
            return redirect("m_transmit_job", pk=pk)
        return redirect("m_preview_job", pk=pk)
    return render(request, "jobs/mobile/preview.html", {
        "job": job, "attachments": _attachment_entries(job), "active_tab": "",
    })


def m_transmit_job(request, pk):
    job = get_object_or_404(Job, pk=pk)
    if request.method == "POST":
        if not (job.email_hr or "").strip():
            messages.error(request, "Fără Kontakt Email.")
            return redirect("m_job_detail", pk=pk)
        live = request.POST.get("live") == "on"

        if not live:
            if job.send_state != "generated":
                messages.error(request, f"Starea e „{job.send_state}” — doar „generated” se transmite.")
                return redirect("m_job_detail", pk=pk)
            res = generation_bridge.send_package(job, live=False)
            if res["ok"]:
                messages.info(request, "Dry-run OK — nimic trimis.")
            else:
                job.send_error = res["error"]
                job.save(update_fields=["send_error"])
                messages.error(request, f"Dry-run eșuat: {res['error']}")
            return redirect("m_transmit_job", pk=pk)

        verdict = generation_bridge.already_sent_in_sheet(job)
        if verdict == "da":
            messages.error(request, "Sheet-ul arată jobul DEJA trimis — LIVE blocat.")
            return redirect("m_transmit_job", pk=pk)

        claimed = Job.objects.filter(pk=pk, send_state="generated").update(send_state="sending")
        if claimed != 1:
            job.refresh_from_db()
            messages.error(request, f"Trimitere neîncepută: starea e „{job.send_state}”.")
            return redirect("m_job_detail", pk=pk)
        job.refresh_from_db()

        res = generation_bridge.send_package(job, live=True)
        if res["ok"]:
            job.send_state = "sent"
            job.sent_at = timezone.now()
            job.send_error = ""
            job.save(update_fields=["send_state", "sent_at", "send_error"])
            mark = sheet_bridge.mark_sent_in_sheet(job)
            if not mark["ok"]:
                messages.warning(request, f"Email TRIMIS, Sheet NU s-a marcat ({mark['error']}).")
            from .models import Job as _JobModel
            sent_today = _JobModel.objects.filter(
                send_state="sent", sent_at__date=timezone.now().date()
            ).count()
            return render(request, "jobs/mobile/success.html", {
                "job": job, "sent_today": sent_today, "active_tab": "",
            })
        job.send_state = "error"
        job.send_error = res["error"]
        job.save(update_fields=["send_state", "send_error"])
        messages.error(request, f"Trimitere eșuată: {res['error']}")
        return redirect("m_job_detail", pk=pk)
    return render(request, "jobs/mobile/transmit.html", {
        "job": job, "attachments": _attachment_entries(job),
        "sheet_verdict": generation_bridge.already_sent_in_sheet(job),
        "active_tab": "",
    })
```

```python
from . import search_bridge

from .models import SearchRun


def m_search_index(request):
    presets = search_bridge.presets()
    selected_preset_key = ""
    if request.method == "POST":
        form = SearchForm(request.POST)
        if form.is_valid():
            res = search_bridge.start_search(form.cleaned_data)
            if res["ok"]:
                messages.info(request, "Căutare pornită.")
                return redirect("m_search_run", pk=res["run"].pk)
            messages.error(request, res["error"])
            if res["run"]:
                return redirect("m_search_run", pk=res["run"].pk)
        else:
            messages.error(request, "Formular invalid.")
    else:
        search_bridge.expire_stale_runs()
        requested_key = request.GET.get("preset")
        selected = next((p for p in presets if p["key"] == requested_key), None)
        if selected is None:
            selected = presets[0] if presets else None
        selected_preset_key = selected["key"] if selected else ""
        form = SearchForm(initial=selected)
    return render(request, "jobs/mobile/search.html", {
        "form": form, "presets": presets, "selected_preset_key": selected_preset_key,
        "runs": SearchRun.objects.all()[:20], "active_tab": "search",
    })


def m_search_run(request, pk):
    run = get_object_or_404(SearchRun, pk=pk)
    if run.status == SearchRun.RUNNING:
        search_bridge.expire_stale_runs()
        run.refresh_from_db()
    order = "-prescore" if request.GET.get("sort") != "data" else "-date_posted"
    return render(request, "jobs/mobile/search_run.html", {
        "run": run, "results": run.results.order_by(order, "pk"),
        "is_running": run.status == SearchRun.RUNNING, "active_tab": "search",
    })


def m_import_search_results(request, pk):
    if request.method != "POST":
        return HttpResponseNotAllowed(["POST"])
    run = get_object_or_404(SearchRun, pk=pk)
    ids = [int(i) for i in request.POST.getlist("result_ids") if str(i).isdigit()]
    ids = list(run.results.filter(pk__in=ids).values_list("pk", flat=True))
    report = search_bridge.import_results(ids)
    if report["errors"]:
        messages.warning(request, report["message"])
    elif report["created"] or report["updated"]:
        messages.success(request, report["message"])
    elif report["skipped"]:
        messages.warning(request, report["message"])
    else:
        messages.error(request, report["message"])
    return redirect("m_search_run", pk=pk)
```

```python
"""Citire + scriere config/settings.yaml pentru ecranul Settings mobil.
Backup .bak la fiecare scriere — fișierul e critic pentru tot pipeline-ul."""
from pathlib import Path

import yaml

_DEFAULT_PATH = Path(__file__).resolve().parents[2] / "config" / "settings.yaml"


def _load(path: Path) -> dict:
    return yaml.safe_load(path.read_text()) or {}


def read_settings(path: Path | None = None) -> dict:
    path = path or _DEFAULT_PATH
    raw = _load(path)
    markets_cfg = ((raw.get("scraping") or {}).get("markets") or {})
    markets = [
        {
            "key": key,
            "enabled": m.get("enabled", True),
            "language": m.get("language", ""),
            "remote": m.get("remote", False),
        }
        for key, m in markets_cfg.items()
    ]
    email_cfg = raw.get("email") or {}
    return {
        "provider": email_cfg.get("active_provider", "gmx"),
        "max_emails_per_day": email_cfg.get("max_emails_per_day", 10),
        "min_prescore": ((raw.get("scraping") or {}).get("filters") or {}).get("min_prescore", 10),
        "markets": markets,
    }


def write_settings(updates: dict, path: Path | None = None) -> dict:
    path = path or _DEFAULT_PATH
    try:
        raw = _load(path)
        backup = path.with_suffix(path.suffix + ".bak")
        backup.write_text(path.read_text())

        if "active_provider" in updates:
            raw.setdefault("email", {})["active_provider"] = updates["active_provider"]
        if "max_emails_per_day" in updates:
            raw.setdefault("email", {})["max_emails_per_day"] = updates["max_emails_per_day"]
        if "market_toggles" in updates:
            markets = raw.setdefault("scraping", {}).setdefault("markets", {})
            for key, enabled in updates["market_toggles"].items():
                if key in markets:
                    markets[key]["enabled"] = enabled

        path.write_text(yaml.safe_dump(raw, allow_unicode=True, sort_keys=False))
        return {"ok": True}
    except (OSError, yaml.YAMLError) as e:
        return {"ok": False, "error": str(e)}
```

```python
class MobileLoginRequiredMiddleware:
```
